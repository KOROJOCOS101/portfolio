2. SCHEMATIC DATABASE ARCHITECTURE DDL (PostgreSQL)
------------------------------------------------------------------------
Execute the following statements sequentially to build out the backend relations:

-- Create Menu Items 
CREATE TABLE menu_items (
    menu_item_id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    category VARCHAR(50) NOT NULL,
    description TEXT,
    menu_price DECIMAL(10,2) NOT NULL,
    menu_quantity INTEGER DEFAULT 0,
    image TEXT
);

-- Create Users 
CREATE TABLE users (
    user_id SERIAL PRIMARY KEY,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    password TEXT NOT NULL,
    user_sign_up TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Create Orders 
CREATE TABLE orders (
    order_id SERIAL PRIMARY KEY,
    user_id INTEGER REFERENCES users(user_id),
    total DECIMAL(10,2) NOT NULL,
    order_placed TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Create Order Items 
CREATE TABLE order_items (
    order_item_id SERIAL PRIMARY KEY,
    order_id INTEGER REFERENCES orders(order_id) ON DELETE CASCADE,
    menu_item_id INTEGER REFERENCES menu_items(menu_item_id) ON DELETE RESTRICT,
    order_quantity INTEGER NOT NULL,
    order_price DECIMAL(10,2) NOT NULL
);
